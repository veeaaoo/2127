#include <stdio.h>
#include <stdlib.h>
#include <time.h>

int main() 
{
    int n, min, max;
    float arr[100];
    float sum = 0;

    printf("Vvedite razmer massiva n (do 100): ");
    scanf_s("%d", &n);

    printf("Vvedite diapazon (min and max): ");
    scanf_s("%d %d", &min, &max);

    srand((unsigned int)time(NULL));

    printf("Sgenerirovanny massiv:\n");
    for (int i = 0; i < n; i++) {
        arr[i] = min + (float)rand() / RAND_MAX * (max - min);
        printf("%.2f ", arr[i]);
    }
    printf("\n");

    for (int i = 0; i < n; i++) {
        int to_subtract = 0;

        for (int j = 0; j < n; j++) {
            int frac_part = (int)((arr[j] - (int)arr[j]) * 10);
            if (frac_part < 0) 
            {
                frac_part = -frac_part;
            }

            if (i == frac_part) 
            {
                to_subtract = 1;
                break;
            }
        }

        if (to_subtract) {
            sum -= arr[i];
        }
        else {
            sum += arr[i];
        }
    }

    printf("Itogovaya summa: %.2f\n", sum);

    return 0;
}
